LECTURA
  SELECT table_name FROM user_tables ORDER BY table_name;
  SELECT view_name FROM user_views ORDER BY view_name;
  SELECT id_deposito, estado_texto FROM vw_deposito_detalle
  ORDER BY id_deposito FETCH FIRST 5 ROWS ONLY;
  SELECT trigger_name, status FROM user_triggers ORDER BY trigger_name;
  SELECT sequence_name FROM user_sequences ORDER BY sequence_name;
  SELECT index_name, uniqueness FROM user_indexes WHERE table_name='DEPOSITO';
  SELECT object_name,object_type,status FROM user_objects
  WHERE object_type IN ('PACKAGE','PACKAGE BODY','PROCEDURE','FUNCTION');
  SELECT name,type,line,text FROM user_errors ORDER BY name,type,line;

EXEC se usa para llamadas PL/SQL cortas. PRINT lee enlaces y cursores.
