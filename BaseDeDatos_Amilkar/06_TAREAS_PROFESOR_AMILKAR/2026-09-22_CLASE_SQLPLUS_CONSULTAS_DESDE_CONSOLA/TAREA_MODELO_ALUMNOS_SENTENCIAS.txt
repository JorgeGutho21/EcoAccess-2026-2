REM =====================================================================
REM 7. CONSULTAS JOIN - MOSTRAR LAS RELACIONES DEL MODELO
REM    Funcionan aunque no haya FK porque se comparan los valores.
REM =====================================================================

PROMPT.
PROMPT [7.1] FACULTAD con sus CARRERAS
SELECT f.nombre_facultad,
       c.nombre_carrera,
       c.creditos
FROM facultad f
JOIN carrera c
  ON f.cod_facultad = c.cod_facultad
ORDER BY f.nombre_facultad, c.nombre_carrera;

PROMPT.
PROMPT [7.2] CARRERA con sus MATERIAS
SELECT c.nombre_carrera,
       m.nombre_materia
FROM carrera c
JOIN materia m
  ON c.cod_carrera = m.cod_carrera
ORDER BY c.nombre_carrera, m.nombre_materia;

PROMPT.
PROMPT [7.3] FACULTAD -> CARRERA -> MATERIA
SELECT f.nombre_facultad,
       c.nombre_carrera,
       m.nombre_materia
FROM facultad f
JOIN carrera c
  ON f.cod_facultad = c.cod_facultad
JOIN materia m
  ON c.cod_carrera = m.cod_carrera
ORDER BY f.nombre_facultad, c.nombre_carrera, m.nombre_materia;

PROMPT.
PROMPT [7.4] DOCENTES y MATERIAS que dictan
SELECT d.nombre_docente,
       m.nombre_materia
FROM docente d
JOIN docente_materia dm
  ON d.id_docente = dm.id_docente
JOIN materia m
  ON dm.cod_materia = m.cod_materia
ORDER BY d.nombre_docente, m.nombre_materia;

PROMPT.
PROMPT [7.5] Cantidad de DOCENTES por MATERIA
SELECT m.nombre_materia,
       COUNT(*) AS cantidad_docentes
FROM materia m
JOIN docente_materia dm
  ON m.cod_materia = dm.cod_materia
GROUP BY m.nombre_materia
ORDER BY m.nombre_materia;

PROMPT.
PROMPT [7.6] DOCENTES y sus NACIONALIDADES
SELECT d.nombre_docente,
       dn.nacionalidad
FROM docente d
JOIN docente_nacionalidad dn
  ON d.id_docente = dn.id_docente
ORDER BY d.nombre_docente, dn.nacionalidad;
